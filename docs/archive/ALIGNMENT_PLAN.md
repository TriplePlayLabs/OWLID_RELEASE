# OwlID Architecture Alignment Plan

**Date:** 2025-11-18
**Purpose:** Align OwlID with MyIdentityNight architecture, focusing on WASM SDK and proof system correctness

---

## Executive Summary

After thorough analysis of both codebases:

**OwlID Current State:**

- ✅ Strong backend proof system (Merkle trees, Ed25519, selective disclosure)
- ✅ REST API verification service with PostgreSQL persistence
- ✅ React frontend with three-role dashboard
- ❌ **NO WASM SDK** - All crypto operations are server-side
- ❌ No client-side proof generation
- ❌ No blockchain/smart contract integration
- ❌ Keys stored in localStorage (plaintext, not encrypted)

**Critical Privacy Issue:** Users must send credentials to server for proof generation, violating zero-knowledge principles.

**Example Rust Implementation with napi-rs:**

```rust
// sdk/owl-id/src/lib.rs
use napi::bindgen_prelude::*;
use napi_derive::napi;

#[napi]
pub struct KeyPair {
    inner: owl_crypto::KeyPair,
}

#[napi]
impl KeyPair {
    #[napi(factory)]
    pub fn generate() -> Self {
        KeyPair {
            inner: owl_crypto::KeyPair::generate(),
        }
    }

    #[napi(factory)]
    pub fn from_hex(private_key: String) -> Result<Self> {
        let inner = owl_crypto::KeyPair::from_hex(&private_key)
            .map_err(|e| Error::from_reason(e.to_string()))?;
        Ok(KeyPair { inner })
    }

    #[napi]
    pub fn public_key(&self) -> PublicKey {
        PublicKey {
            inner: self.inner.public_key(),
        }
    }

    #[napi]
    pub fn private_key_hex(&self) -> String {
        self.inner.private_key_hex()
    }

    #[napi]
    pub fn sign(&self, message: Buffer) -> Signature {
        let sig = self.inner.sign(message.as_ref());
        Signature { inner: sig }
    }
}

#[napi(object)]
pub struct InformationRequest {
    pub mandatory: Vec<String>,
    pub optional: Vec<String>,
    pub trusted_issuers: Vec<String>,
    pub challenge: String,
}

#[napi(object)]
pub struct VerificationResult {
    pub success: bool,
    pub reason: Option<String>,
    pub disclosed_attributes: Option<serde_json::Value>,
}

#[napi]
pub fn generate_key_pair() -> KeyPair {
    KeyPair::generate()
}

#[napi]
pub fn sha256(data: Buffer) -> String {
    use sha2::{Sha256, Digest};
    let hash = Sha256::digest(data.as_ref());
    hex::encode(hash)
}

#[napi]
pub fn blake3(data: Buffer) -> String {
    let hash = blake3::hash(data.as_ref());
    hash.to_hex().to_string()
}

// frontend/src/lib/secure-storage.ts

interface EncryptedKeyStore {
  version: number
  encrypted_keys: string // AES-GCM encrypted JSON
  salt: string // For PBKDF2 (hex)
  iv: string // Initialization vector (hex)
}

interface KeyData {
  id: string
  label: string
  public_key: string
  private_key: string
  type: 'issuer' | 'owner'
  created_at: string
}

// Derive encryption key from password using PBKDF2
async function deriveKey(password: string, salt: Uint8Array): Promise<CryptoKey> {
  const passwordKey = await crypto.subtle.importKey(
    'raw',
    new TextEncoder().encode(password),
    'PBKDF2',
    false,
    ['deriveKey'],
  )

  return crypto.subtle.deriveKey(
    {
      name: 'PBKDF2',
      salt: salt,
      iterations: 100000,
      hash: 'SHA-256',
    },
    passwordKey,
    { name: 'AES-GCM', length: 256 },
    false,
    ['encrypt', 'decrypt'],
  )
}

// Encrypt data with AES-GCM
async function encrypt(key: CryptoKey, data: string, iv: Uint8Array): Promise<string> {
  const encrypted = await crypto.subtle.encrypt(
    { name: 'AES-GCM', iv },
    key,
    new TextEncoder().encode(data),
  )

  return bytesToHex(new Uint8Array(encrypted))
}

// Decrypt data with AES-GCM
async function decrypt(key: CryptoKey, encryptedHex: string, iv: Uint8Array): Promise<string> {
  const decrypted = await crypto.subtle.decrypt(
    { name: 'AES-GCM', iv },
    key,
    hexToBytes(encryptedHex),
  )

  return new TextDecoder().decode(decrypted)
}

// Store encrypted keys
export async function storeKeys(keys: KeyData[], password: string): Promise<void> {
  const salt = crypto.getRandomValues(new Uint8Array(16))
  const iv = crypto.getRandomValues(new Uint8Array(12))

  const derivedKey = await deriveKey(password, salt)
  const encryptedData = await encrypt(derivedKey, JSON.stringify(keys), iv)

  const store: EncryptedKeyStore = {
    version: 1,
    encrypted_keys: encryptedData,
    salt: bytesToHex(salt),
    iv: bytesToHex(iv),
  }

  localStorage.setItem('owlid_encrypted_keys', JSON.stringify(store))
}

// Load and decrypt keys
export async function loadKeys(password: string): Promise<KeyData[]> {
  const storedData = localStorage.getItem('owlid_encrypted_keys')
  if (!storedData) {
    throw new Error('No encrypted keys found')
  }

  const store: EncryptedKeyStore = JSON.parse(storedData)

  const derivedKey = await deriveKey(password, hexToBytes(store.salt))
  const decryptedJson = await decrypt(derivedKey, store.encrypted_keys, hexToBytes(store.iv))

  return JSON.parse(decryptedJson)
}

// Utility functions
function bytesToHex(bytes: Uint8Array): string {
  return Array.from(bytes)
    .map((b) => b.toString(16).padStart(2, '0'))
    .join('')
}

function hexToBytes(hex: string): Uint8Array {
  const bytes = new Uint8Array(hex.length / 2)
  for (let i = 0; i < hex.length; i += 2) {
    bytes[i / 2] = parseInt(hex.substr(i, 2), 16)
  }
  return bytes
}
```

**Verification Flow:**

```typescript
// frontend/src/lib/verification.ts
import { Token, PublicKey } from 'owl-id'

interface CachedIssuer {
  public_key: string
  name: string
  cached_at: number
}

async function getCachedTrustedIssuers(): Promise<PublicKey[]> {
  const db = await openDB('owlid_cache', 1)
  const cached: CachedIssuer[] = await db.getAll('trusted_issuers')
  return cached.map((i) => PublicKey.from_hex(i.public_key))
}

async function syncTrustedIssuers(): Promise<void> {
  try {
    const response = await fetch(`${VERIFICATION_API_URL}/trusted-issuers`)
    const issuers = await response.json()

    const db = await openDB('owlid_cache', 1)
    const tx = db.transaction('trusted_issuers', 'readwrite')

    for (const issuer of issuers) {
      await tx.store.put({
        public_key: issuer.public_key,
        name: issuer.name,
        cached_at: Date.now(),
      })
    }

    await tx.done
  } catch (error) {
    console.error('Failed to sync trusted issuers:', error)
  }
}

export async function verifyToken(
  tokenJson: string,
  challenge: string,
): Promise<VerificationResult & { offline: boolean }> {
  const token = Token.from_json(tokenJson)

  // 1. Try offline verification first
  const cachedIssuers = await getCachedTrustedIssuers()
  const offlineResult = token.verify(cachedIssuers, challenge)

  if (!offlineResult.success) {
    return { ...offlineResult, offline: true }
  }

  // 2. Check revocation status online (if available)
  try {
    const revocationStatus = await checkRevocationOnline(token.root_hash())

    if (revocationStatus.is_revoked) {
      return {
        success: false,
        reason: 'credential_revoked',
        offline: false,
      }
    }

    return { ...offlineResult, offline: false }
  } catch (error) {
    // Network error, return offline result with warning
    console.warn('Could not check revocation status online:', error)
    return { ...offlineResult, offline: true }
  }
}

async function checkRevocationOnline(rootHash: string): Promise<RevocationStatus> {
  const response = await fetch(`${VERIFICATION_API_URL}/revocations/check`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ root_hash: rootHash }),
  })

  if (!response.ok) {
    throw new Error('Revocation check failed')
  }

  return response.json()
}
```

**Critical:** Ensure proof generation and verification are cryptographically sound

5. **Collision Resistance:** Hash function must be collision-resistant (SHA-256 ✓)

// contracts/IssuerRegistry.sol
pragma solidity ^0.8.0;

contract IssuerRegistry {
    struct Issuer {
        string publicKey;     // Ed25519 public key (hex)
        string name;          // Issuer name
        bool active;          // Can be deactivated
        uint256 registeredAt;
    }

    mapping(string => Issuer) public issuers;
    string[] public issuerKeys;
    address public owner;

    event IssuerRegistered(string publicKey, string name);
    event IssuerDeactivated(string publicKey);

    constructor() {
        owner = msg.sender;
    }

    modifier onlyOwner() {
        require(msg.sender == owner, "Only owner can call");
        _;
    }

    function registerIssuer(
        string memory publicKey,
        string memory name
    ) public onlyOwner {
        require(bytes(issuers[publicKey].publicKey).length == 0, "Already registered");

        issuers[publicKey] = Issuer({
            publicKey: publicKey,
            name: name,
            active: true,
            registeredAt: block.timestamp
        });

        issuerKeys.push(publicKey);
        emit IssuerRegistered(publicKey, name);
    }

    function deactivateIssuer(string memory publicKey) public onlyOwner {
        require(issuers[publicKey].active, "Not active");
        issuers[publicKey].active = false;
        emit IssuerDeactivated(publicKey);
    }

    function isActiveIssuer(string memory publicKey) public view returns (bool) {
        return issuers[publicKey].active;
    }

    function getAllActiveIssuers() public view returns (Issuer[] memory) {
        uint256 activeCount = 0;
        for (uint256 i = 0; i < issuerKeys.length; i++) {
            if (issuers[issuerKeys[i]].active) {
                activeCount++;
            }
        }

        Issuer[] memory active = new Issuer[](activeCount);
        uint256 index = 0;
        for (uint256 i = 0; i < issuerKeys.length; i++) {
            if (issuers[issuerKeys[i]].active) {
                active[index] = issuers[issuerKeys[i]];
                index++;
            }
        }

        return active;
    }
}

**Example: Revocation Contract:**

```solidity
// contracts/RevocationRegistry.sol
pragma solidity ^0.8.0;

contract RevocationRegistry {
    enum Status { Active, Revoked, Suspended }

    struct RevocationEntry {
        Status status;
        uint256 timestamp;
        string reason;
        string issuerKey;
    }

    mapping(string => RevocationEntry) public revocations;

    event CredentialRevoked(string rootHash, string reason);
    event CredentialSuspended(string rootHash);
    event CredentialReactivated(string rootHash);

    // Only issuer who issued the credential can revoke it
    modifier onlyIssuer(string memory rootHash, string memory issuerKey) {
        require(
            keccak256(bytes(revocations[rootHash].issuerKey)) == keccak256(bytes(issuerKey)),
            "Not the issuer"
        );
        _;
    }

    function revoke(
        string memory rootHash,
        string memory issuerKey,
        string memory reason
    ) public {
        revocations[rootHash] = RevocationEntry({
            status: Status.Revoked,
            timestamp: block.timestamp,
            reason: reason,
            issuerKey: issuerKey
        });

        emit CredentialRevoked(rootHash, reason);
    }

    function suspend(string memory rootHash, string memory issuerKey)
        public
        onlyIssuer(rootHash, issuerKey)
    {
        revocations[rootHash].status = Status.Suspended;
        emit CredentialSuspended(rootHash);
    }

    function reactivate(string memory rootHash, string memory issuerKey)
        public
        onlyIssuer(rootHash, issuerKey)
    {
        revocations[rootHash].status = Status.Active;
        emit CredentialReactivated(rootHash);
    }

    function isRevoked(string memory rootHash) public view returns (bool) {
        return revocations[rootHash].status == Status.Revoked ||
               revocations[rootHash].status == Status.Suspended;
    }
}
```
